using System.Collections; // 실제 탐사 화면 대기 사용
using NUnit.Framework; // 탐사 화면 구성 검증 사용
using TMPro; // 실제 재화 숫자 확인 사용
using UnityEngine.UI; // 실제 상점 닫기 버튼 사용
using UnityEngine; // 실제 탐사 화면 객체 사용
using UnityEngine.SceneManagement; // 실제 씬 전환 사용
using UnityEngine.TestTools; // 실제 플레이 모드 검사 사용
public sealed class ExplorationResourceHudTests // 탐사 화면 우측 하단 정리 검사
{ // 탐사 화면 정리 검사 시작
    [UnityTest] // 실제 탐사 화면 검사 지정
    public IEnumerator ResourcesUpdateAndShopRoomRemainsAccessible() // 시험 정보와 저장 버튼 제거 및 탐사 유지 확인
    { // 실제 탐사 화면 검사 시작
        yield return new EnterPlayMode(); // 실제 게임 플레이 시작
        SceneManager.LoadScene("00_Boot"); // 실제 게임 시작 씬 진입
        for (float end = Time.realtimeSinceStartup + 30f; Time.realtimeSinceStartup < end && Object.FindFirstObjectByType<MainMenuController>() == null;) // 실제 시작 메뉴 대기
        { // 시작 메뉴 대기 시작
            yield return null; // 다음 화면 갱신 대기
        } // 시작 메뉴 대기 종료
        Assert.IsNotNull(Object.FindFirstObjectByType<MainMenuController>()); // 실제 시작 메뉴 확인
        Object.FindFirstObjectByType<MainMenuController>().StartGame(); // 실제 거점 진입
        for (float end = Time.realtimeSinceStartup + 30f; Time.realtimeSinceStartup < end && Object.FindFirstObjectByType<LobbyLayoutView>() == null;) // 실제 거점 대기
        { // 거점 대기 시작
            yield return null; // 다음 화면 갱신 대기
        } // 거점 대기 종료
        Assert.IsNotNull(Object.FindFirstObjectByType<LobbyLayoutView>()); // 실제 거점 확인
        PartyDepartureTestSupport.StartExploration(0, 1, 2, 3); // 실제 네 명 파티 탐사 시작
        for (float end = Time.realtimeSinceStartup + 30f; Time.realtimeSinceStartup < end && Object.FindFirstObjectByType<ExplorationPlayerController>() == null;) // 실제 탐사 화면 대기
        { // 탐사 화면 대기 시작
            yield return null; // 다음 화면 갱신 대기
        } // 탐사 화면 대기 종료
        Assert.IsNotNull(Object.FindFirstObjectByType<ExplorationPlayerController>()); // 실제 탐사 플레이어 확인
        yield return new WaitForSecondsRealtime(0.1f); // 탐사 HUD 생성 완료 대기
        Assert.IsNotNull(Object.FindFirstObjectByType<ExplorationMapRuntime>()); // 실제 탐사 맵 런타임 유지 확인
        Assert.IsNotNull(Object.FindFirstObjectByType<ExplorationTilemapView>()); // 실제 탐사 타일맵 유지 확인
        Assert.IsNull(Object.FindFirstObjectByType<ExplorationMapDebugView>()); // 우측 하단 시험 정보와 미니맵 제거 확인
        Assert.IsNull(Object.FindFirstObjectByType<SaveMenuEntry>()); // 탐사 화면 우측 하단 저장 버튼 제거 확인
        Assert.IsNotNull(Object.FindFirstObjectByType<ExplorationRunLoopDebugView>()); // 탐사 결과와 거점 복귀 처리 유지 확인
        ExplorationResourceHud hud = Object.FindFirstObjectByType<ExplorationResourceHud>(); // 실제 탐사 재화 화면 조회
        Assert.IsNotNull(hud); // 실제 탐사 재화 화면 확인
        Assert.AreEqual(new Vector2(1f, 1f), (hud.transform as RectTransform).anchorMax); // 재화의 화면 오른쪽 위 배치 확인
        PlayerResourceManager resources = PlayerResourceManager.EnsureInstance(); // 실제 재화 관리자 조회
        resources.ResetResources(); // 독립 테스트 재화 초기화
        resources.AddResources(123, 45, 67, 89); // 실제 네 재화 획득
        string[] names = { "Gold", "Screw", "Iron", "Wire" }; // 실제 재화 아이콘 순서
        string[] expected = { "123", "45", "67", "89" }; // 실제 획득 재화 숫자
        for (int index = 0; index < names.Length; index++) // 네 재화 표시 순회
        { // 개별 재화 표시 검사 시작
            Transform entry = hud.transform.Find(names[index]); // 실제 재화 표시 영역 조회
            Assert.IsNotNull(entry); // 실제 재화 표시 영역 확인
            Assert.IsNotNull(entry.Find("Icon").GetComponent<Image>().sprite); // 제작한 실제 재화 아이콘 확인
            Assert.AreEqual(expected[index], entry.Find("Value").GetComponent<TMP_Text>().text); // 재화 이름 없이 실제 숫자 표시 확인
        } // 개별 재화 표시 검사 종료
        Assert.IsTrue(resources.TrySpend(23, 5, 7, 9)); // 실제 네 재화 소비 확인
        Assert.AreEqual("100", hud.transform.Find("Gold/Value").GetComponent<TMP_Text>().text); // 골드 소비 즉시 갱신 확인
        Assert.AreEqual("40", hud.transform.Find("Screw/Value").GetComponent<TMP_Text>().text); // 나사 소비 즉시 갱신 확인
        Assert.IsNull(GameObject.Find("OpenShopButton")); // 화면 우측 상점 시험 버튼 제거 확인
        TMP_Text progress = GameObject.Find("PersistentProgress").GetComponent<TMP_Text>(); // 실제 탐사 진행 글자 조회
        Assert.IsFalse(progress.text.Contains("호감도")); // 호감도 문구 제거 확인
        Assert.IsFalse(progress.text.Contains("이번 탐사 클리어")); // 클리어 조우 문구 제거 확인
        Assert.IsFalse(progress.text.Contains("Gold") || progress.text.Contains("나사") || progress.text.Contains("철판") || progress.text.Contains("전선")); // 재화 이름 글자 제거 확인
        ShopPrototypeView shop = Object.FindFirstObjectByType<ShopPrototypeView>(); // 실제 상점 화면 조회
        Assert.IsNotNull(shop); // 실제 상점 화면 유지 확인
        ExplorationMapRuntime map = Object.FindFirstObjectByType<ExplorationMapRuntime>(); // 실제 탐사 맵 조회
        GameObject roomObject = new GameObject("TestShopRoom", typeof(ExplorationSpecialRoomView)); // 실제 상점 방 상호작용 생성
        ExplorationSpecialRoomView room = roomObject.GetComponent<ExplorationSpecialRoomView>(); // 실제 상점 방 화면 조회
        room.Initialize(map, Vector2Int.zero, ExplorationRoomType.Shop, "resource_hud_test_shop"); // 실제 상점 방 역할과 고유 식별자 설정
        ExplorationPlayerController player = Object.FindFirstObjectByType<ExplorationPlayerController>(); // 실제 탐사 플레이어 조회
        room.SendMessage("OnTriggerEnter2D", player.GetComponent<Collider2D>()); // 실제 플레이어의 상점 방 진입 처리
        Assert.IsTrue(shop.transform.Find("ShopOverlay").gameObject.activeSelf); // 상점 방을 통한 실제 상점 열기 확인
        Assert.IsTrue(ExplorationSessionManager.Instance.IsEventResolved("resource_hud_test_shop")); // 실제 상점 방 방문 처리 확인
        shop.transform.Find("ShopOverlay/ShopPanel/CloseButton").GetComponent<Button>().onClick.Invoke(); // 실제 상점 닫기
        Object.DestroyImmediate(roomObject); // 임시 상점 방 정리
        yield return new ExitPlayMode(); // 실제 게임 플레이 종료
    } // 실제 탐사 화면 검사 종료
} // 탐사 화면 정리 검사 종료
